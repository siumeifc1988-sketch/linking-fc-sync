import os import json import time from curl_cffi import requests as curl_requests import requests as py_requests CLUB_ID = "41026" TYPES = { "leagueMatch": "聯賽", "friendlyMatch": "友誼賽", "playoffMatch": "季後賽" } headers = { "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/124.0.0.0 Safari/537.36", "Accept": "application/json", "Referer": "https://www.ea.com/", "Origin": "https://www.ea.com" } all_matches = [] for match_type in TYPES: url = f"https://proclubs.ea.com/api/fc/clubs/matches?platform=common-gen5&clubIds={CLUB_ID}&matchType={match_type}&maxResultCount=20" try: r = curl_requests.get(url, headers=headers, impersonate="chrome124", timeout=30) print(f"{match_type}: {r.status_code}") if r.status_code == 200: data = r.json() for m in data: m["_matchType"] = match_type all_matches.append(m) print(f"{match_type} 捉到 {len(data)} 場") except Exception as e: print(f"{match_type} 錯誤 {e}") time.sleep(2) dedup = {str(m['matchId']): m for m in all_matches} merged = sorted(dedup.values(), key=lambda x: int(x.get('timestamp', 0)), reverse=True) # ---------- 球員表現 ---------- players_raw = [] players_stats = [] def fetch_players(endpoint): try: r = curl_requests.get(endpoint, headers=headers, impersonate="chrome124", timeout=30) print(f"{endpoint} -> {r.status_code}") if r.status_code == 200: return r.json() except Exception as e: print(f"球員API錯誤 {e}") return None # EA 有幾個不同路徑，試哂 urls_members = [ f"https://proclubs.ea.com/api/fc/clubs/{CLUB_ID}/members?platform=common-gen5", f"https://proclubs.ea.com/api/fc/members/stats?platform=common-gen5&clubId={CLUB_ID}", f"https://proclubs.ea.com/api/fc/clubs/{CLUB_ID}/members/stats?platform=common-gen5" ] members_list = None members_stats = None for u in urls_members: j = fetch_players(u) if j: if isinstance(j, dict) and "members" in j: members_list = j["members"] elif isinstance(j, list) and len(j)>0 and "name" in str(j[0]).lower(): # 可能是 members if "gamesPlayed" not in str(j[0]): members_list = j else: members_stats = j elif isinstance(j, list): members_stats = j time.sleep(1) # 嘗試第二組專門 stats if not members_stats: j = fetch_players(f"https://proclubs.ea.com/api/fc/clubs/{CLUB_ID}/members/stats?platform=common-gen5") if j: members_stats = j if isinstance(j, list) else j.get("members",[]) print(f"members: {len(members_list) if members_list else 0} stats: {len(members_stats) if members_stats else 0}") # 合併 players_map = {} if members_list: for m in members_list: pid = str(m.get("playerId") or m.get("id") or m.get("name")) players_map[pid] = {"playerId": pid, "name": m.get("name") or m.get("playerName") or "球員", "proPos": m.get("favoritePosition") or m.get("pos") or "", "raw": m} if members_stats: for s in members_stats: pid = str(s.get("playerId") or s.get("id") or s.get("name")) if pid not in players_map: players_map[pid] = {"playerId": pid, "name": s.get("name") or s.get("playerName") or "球員"} # 合併 stats players_map[pid].update(s) final_players = list(players_map.values()) # 按入球排序 final_players.sort(key=lambda x: int(x.get("goals",0) or 0), reverse=True) os.makedirs("data", exist_ok=True) with open("data/matches.json", "w", encoding="utf-8") as f: json.dump({"matches": merged, "updated": int(time.time()), "count": len(merged)}, f, ensure_ascii=False, indent=2) with open("data/players.json", "w", encoding="utf-8") as f: json.dump({"players": final_players, "updated": int(time.time()), "count": len(final_players)}, f, ensure_ascii=False, indent=2) print(f"寫入 {len(merged)} 場 到 data/matches.json") print(f"寫入 {len(final_players)} 球員 到 data/players.json") # ---------- Firebase ---------- API_KEY = os.getenv("FIREBASE_API_KEY") PROJECT = os.getenv("FIREBASE_PROJECT", "football-signup-64bd9") if not API_KEY: print("無 FIREBASE_API_KEY，跳過 Firebase") else: # matches for m in merged: match_id = str(m.get("matchId")) ts = int(m.get("timestamp", 0)) date_str = time.strftime("%Y-%m-%d %H:%M:%S", time.gmtime(ts)) if ts else "" payload = { "fields": { "matchId": {"stringValue": match_id}, "timestamp": {"integerValue": str(ts)}, "dateStr": {"stringValue": date_str}, "matchType": {"stringValue": m.get("_matchType","")}, "raw": {"stringValue": json.dumps(m, ensure_ascii=False)[:900000]} } } url = f"https://firestore.googleapis.com/v1/projects/{PROJECT}/databases/(default)/documents/matches/{match_id}?key={API_KEY}" try: py_requests.patch(url, json=payload, timeout=15) except: pass time.sleep(0.15) # players for p in final_players: pid = str(p.get("playerId") or p.get("id")) payload = { "fields": { "playerId": {"stringValue": pid}, "name": {"stringValue": p.get("name","")}, "gamesPlayed": {"integerValue": str(int(p.get("gamesPlayed",0) or p.get("matches",0) or 0))}, "goals": {"integerValue": str(int(p.get("goals",0) or 0))}, "assists": {"integerValue": str(int(p.get("assists",0) or 0))}, "rating": {"doubleValue": float(p.get("ratingAve",0) or p.get("rating",0) or 0)}, "raw": {"stringValue": json.dumps(p, ensure_ascii=False)[:900000]} } } url = f"https://firestore.googleapis.com/v1/projects/{PROJECT}/databases/(default)/documents/players/{pid}?key={API_KEY}" try: r = py_requests.patch(url, json=payload, timeout=15) print(f"Firebase player {pid} {p.get('name')} -> {r.status_code}") except Exception as e: print(f"player {pid} fail {e}")
